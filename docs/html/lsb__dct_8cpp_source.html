16void apply_dct_quantization(cv::Mat& block, const int quant_table[8][8]) {
17 block.convertTo(block, CV_32F);
18 cv::dct(block, block);

31size_t embed_lsb_dct(const std::string& image_path,
32 const std::vector<uint8_t> message_bits,
33 const std::string& output_path)
34{
35 cv::Mat img = cv::imread(image_path, cv::IMREAD_GRAYSCALE);
36 if (img.empty()) {
37 std::cerr << "Ошибка: не удалось открыть контейнерное изображение: "
38 << image_path << std::endl;
39 return 0;
40 }
41
42 size_t bit_index = 0;
43 size_t total_bits = message_bits.size();
44
45 for (int y = 0; y + BLOCK_SIZE <= img.rows && bit_index < total_bits; y += BLOCK_SIZE) {
46 for (int x = 0; x + BLOCK_SIZE <= img.cols && bit_index < total_bits; x += BLOCK_SIZE) {
47
48 cv::Mat block = img(cv::Rect(x, y, BLOCK_SIZE, BLOCK_SIZE)).clone();
49 apply_dct_quantization(block, QUANT_TABLE);
50
51 int coeff = static_cast<int>(block.at<float>(2, 2));
52 coeff = (coeff & ~1) | (message_bits[bit_index] & 1);
53 block.at<float>(2, 2) = static_cast<float>(coeff);
54 ++bit_index;
55
56 apply_idct_dequantization(block, QUANT_TABLE);
57 block.copyTo(img(cv::Rect(x, y, BLOCK_SIZE, BLOCK_SIZE)));
58 }
59 }
60
61 if (!cv::imwrite(output_path, img)) {
62 std::cerr << "Ошибка: не удалось сохранить стего-изображение: "
63 << output_path << std::endl;
64 }
65
66 return total_bits;
67}
